# Y Download dependencies
#
# NOTE: ffmpeg and node/deno are NOT pip packages. They are system binaries:
#   ffmpeg    - merges video+audio, converts to MP3
#   node/deno - solves YouTube's signature (nsig) challenges
flask>=3.0

# Keep this recent: YouTube breaks older releases. Update with:
#   .venv/bin/pip install -U yt-dlp
yt-dlp>=2024.1.1

# Official companion package: bundles the JS challenge solver (EJS) that
# YouTube's signature protection needs. Without it yt-dlp downloads the solver
# from GitHub at runtime - fine locally, fragile inside a container.
yt-dlp-ejs>=0.3.0

# TLS/JA3 browser impersonation. Many sites answer a datacenter IP with a
# Cloudflare 403; yt-dlp needs this to work around it. Without it the app
# simply reports the 403 rather than crashing (see ytdlp_base_opts).
curl-cffi>=0.7.0

# Production WSGI server (used by the Docker image / Render).
gunicorn>=21.2
